
#include <stdio.h>
#define MAX 5

int queue[MAX];
int front = -1, rear = -1;

// Kuyruğun durumunu yazdır
void printQueueState() {
    printf("Durum: ");
    for (int i = 0; i < MAX; i++) {
        printf("queue[%d] = ", i);
        if (front == -1) //kuyruk tamamen boş demek
            printf("empty  ");
        else if (front <= rear && (i >= front && i <= rear)) //dümdüz ilerliyor
            printf("%d  ", queue[i]);
        else if (rear < front && (i >= front || i <= rear)) //queue dolmuş da rear başa gelmiş
            printf("%d  ", queue[i]);
        else
            printf("empty  ");
    }
    printf("\n\n");
}

// Eleman ekleme (enqueue)
void enqueue(int value) {
    if ((rear + 1) % MAX == front) {
        printf("Queue dolu! Eleman eklenemedi: %d\n\n", value);
        return;
    }

    if (front == -1) // ilk eleman ekleniyorsa
        front = 0;

    rear = (rear + 1) % MAX;
    queue[rear] = value;

    printf("ENQUEUE -> %d eklendi (front=%d, rear=%d)\n", value, front, rear);
    printQueueState();
}

// Eleman çıkarma (dequeue)
void dequeue() {
    if (front == -1) {
        printf("Queue bos! Eleman cikarilamadi.\n\n");
        return;
    }

    printf("DEQUEUE -> %d cikarildi (front=%d)\n", queue[front], front);

    if (front == rear) { // son eleman da çıktıysa
        front = rear = -1;
    } else {
        front = (front + 1) % MAX;
    }

    printQueueState();
}

int main() {
    enqueue(10);
    enqueue(20);
    enqueue(30);
    enqueue(40);
    dequeue();
    enqueue(50);
    enqueue(60);
    dequeue();
    enqueue(70);
    dequeue();
    enqueue(80);
    enqueue(90);
    dequeue();
    dequeue();
    dequeue();
    dequeue();
    return 0;
}
