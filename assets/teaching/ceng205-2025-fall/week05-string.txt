#include <stdio.h>
#include <string.h>
#define MAX 20

char queue[MAX];
int front = -1, rear = -1;

// Eleman ekleme
void enqueue(char c) {
    if ((rear + 1) % MAX == front) {
        printf("Queue dolu!\n");
        return;
    }
    if (front == -1)  // ilk eleman
        front = 0;
    rear = (rear + 1) % MAX;
    queue[rear] = c;
}

// Eleman çıkarma
char dequeue() {
    if (front == -1) {
        printf("Queue bos!\n");
        return '\0';
    }
    char c = queue[front];
    if (front == rear)  // son eleman
        front = rear = -1;
    else
        front = (front + 1) % MAX;
    return c;
}

// Kuyruğu yazdır
void printQueue() {
    if (front == -1) {
        printf("(bos)\n");
        return;
    }
    int i = front;
    while (1) {
        printf("%c", queue[i]);
        if (i == rear)
            break;
        i = (i + 1) % MAX;
    }
    printf("\n");
}

int main() {
    char word[20];
    printf("Bir kelime giriniz: ");
    scanf("%s", word);

    // Başlangıçta tüm karakterleri enqueue ets
    for (int i = 0; i < strlen(word); i++)
        enqueue(word[i]);

    printf("\n--- Metin Dondurme Baladi ---\n");
    for (int step = 1; step <= strlen(word); step++) {
        printQueue();                // mevcut kelimeyi göster
        char c = dequeue();          // ilk harfi çıkar
        enqueue(c);                  // çıkarılan harfi sona ekle
    }

    printf("\n--- Dongu Tamamlandi ---\n");
    return 0;
}
