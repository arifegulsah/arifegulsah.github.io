#include <stdio.h>
#include <stdlib.h>
#include <unistd.h> // sleep() fonksiyonu için

#define SIZE 5

int queue[SIZE];
int front = -1;
int rear = -1;

// Kuyruğun boş olup olmadığını kontrol et
int isEmpty() {
    return (front == -1 || front > rear);
}

// Kuyruğun dolu olup olmadığını kontrol et
int isFull() {
    return (rear == SIZE - 1);
}

// Eleman ekle (enqueue)
void enqueue(int value) {
    if (isFull()) {
        printf("Queue is full! %d could not be added.\n", value);
        return;
    }

    if (front == -1) front = 0; // İlk eleman ekleniyorsa front ayarla
    queue[++rear] = value;

    printf("\n Enqueue: %d added\n", value);
    sleep(1);
}

// Eleman çıkar (dequeue)
void dequeue() {
    if (isEmpty()) {
        printf("Queue is empty, cannot dequeue.\n");
        return;
    }

    printf("\n Dequeue: %d removed\n", queue[front++]);
    sleep(1);
}

// Kuyruğun o anki durumunu yazdır
void displayQueue() {
    if (isEmpty()) {
        printf("\nQueue is currently empty.\n");
        return;
    }

    printf("\n Current Queue: ");
    for (int i = front; i <= rear; i++) {
        printf("%d ", queue[i]);
    }
    printf("\n");
    sleep(1);
}

int main() {
    printf("=== Queue Operations ===\n");

    enqueue(10); displayQueue();
    enqueue(20); displayQueue();
    enqueue(30); displayQueue();
    dequeue();   displayQueue();
    enqueue(40); displayQueue();
    dequeue();   displayQueue();
    enqueue(50); displayQueue();
    enqueue(60); displayQueue();
    dequeue();   displayQueue();

    printf("\n All operations completed.\n");
    return 0;
}
