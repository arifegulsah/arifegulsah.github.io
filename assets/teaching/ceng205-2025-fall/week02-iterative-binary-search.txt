 
#include <stdio.h>

// Binary Search fonksiyonu
// Eğer eleman bulunursa indexini döner, yoksa -1 döner
int binarySearch(int arr[], int n, int target) {
    int left = 0, right = n - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (arr[mid] == target) {
            return mid; // bulundu
        }
        else if (arr[mid] < target) {
            left = mid + 1; // sağ tarafa git
        }
        else {
            right = mid - 1; // sol tarafa git
        }
    }
    return -1; // bulunamadı
}

int main() {
    int arr[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
    int n = sizeof(arr) / sizeof(arr[0]);
    int target;

    printf("Aranacak sayiyi girin: ");
    scanf("%d", &target);

    int result = binarySearch(arr, n, target);

    if (result != -1)
        printf("%d sayisi dizide bulundu. Index: %d\n", target, result);
    else
        printf("%d sayisi dizide bulunamadi.\n", target);

    return 0;
}
